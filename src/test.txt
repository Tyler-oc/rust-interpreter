fun fib(n) {

}

// TEST 2: Closure Allocation (Tests your Resolver & Scoping)
// ---------------------------------------------------------
print "2. Testing Lexical Closures (10,000 iterations)...";
fun makeCounter() {
  var i = 0;
  fun count() {
    i = i + 1;
    return i;
  }
  return count;
}

var counter = makeCounter();
var closureResult = 0;
for (var i = 0; i < 10000; i = i + 1) {
  closureResult = counter();
}
print closureResult; // Should print 10000
